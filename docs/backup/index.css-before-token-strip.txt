@layer base {
  /* Default border colour for any bare `border` utility. Was the themed
     `border-border` token, which resolved to the LIGHT lavender once the
     `dark` class stopped being set — this is the phosphor hairline instead. */
  * {
    @apply border-ph-line;
  }
  html {
    -webkit-text-size-adjust: 100%;
    /* In-page anchors (#how, #faq) land under the 64px sticky header without
       this — the heading you jumped to is the one thing it covers. */
    scroll-padding-top: 5rem;
  }
  @media (prefers-reduced-motion: no-preference) {
    html {
      scroll-behavior: smooth;
    }
  }
  body {
    @apply bg-ph-bg text-ph-ink antialiased;
    font-feature-settings: "cv02", "cv03", "cv04", "cv11";
  }
  /* One focus treatment for the whole product. */
  :focus-visible {
    @apply outline-none ring-2 ring-accent/70 ring-offset-2 ring-offset-canvas;
  }
  ::selection {
    @apply bg-accent-soft text-ink;
  }
}
