/**
 * Stitch "Galaxy Theme" building blocks for the public pages (2026-09-24),

import type { ComponentType, ReactNode } from "react";
import { ArrowRight } from "lucide-react";
import { Reveal } from "@/components/motion/reveal";
import galaxy from "@/assets/stitch/galaxy-hero.jpg";

type IconType = ComponentType<{ size?: number; className?: string }>;

const LABEL_SM = "text-[11px] leading-[14px] tracking-[0.04em] font-semibold";
const LABEL_MD = "text-[13px] leading-4 tracking-[0.02em] font-medium";
const BODY_LG = "text-[16px] leading-6 tracking-[-0.01em]";
const H_LG =
  "font-st-display text-[28px] leading-9 tracking-[-0.02em] font-semibold md:text-[36px] md:leading-[44px] md:tracking-[-0.025em]";
const DISPLAY =
  "font-st-display text-[36px] leading-[44px] tracking-[-0.025em] font-semibold md:text-[48px] md:leading-[56px] md:tracking-[-0.03em] md:font-bold";

/** Page root: void background, Geist, and the fixed-nav offset. */
export function GalaxyPage({ children }: { children: ReactNode }) {
  return (
    <div className="relative min-h-screen bg-[#09080e] font-st-body text-st-ink antialiased selection:bg-st-primary-container selection:text-white">
      {children}
    </div>
  );
}

export function Eyebrow({ children, tone = "violet" }: { children: ReactNode; tone?: "violet" | "cyan" }) {
  const cls = tone === "cyan" ? "border-cyan-400/30 text-cyan-300" : "border-st-primary/30 text-st-primary";
  const dot = tone === "cyan" ? "bg-cyan-300 shadow-[0_0_6px_#67e8f9]" : "bg-st-primary shadow-[0_0_6px_#d2bbff]";
  return (
    <div className={`mb-3 inline-flex items-center gap-2 rounded-full border bg-[#121022] px-3 py-1 ${cls}`}>
      <span className={`h-1.5 w-1.5 rounded-full ${dot}`} />
      <span className={`${LABEL_SM} font-mono uppercase tracking-widest`}>{children}</span>
    </div>
  );
}

/** Link styled as a Stitch CTA pill. `primary` = gradient, `glass` = dark outline. */
export function CtaLink({
  href,
  children,
  variant = "primary",
  icon: Icon,
  className = "",
}: {
  href: string;
  children: ReactNode;
  variant?: "primary" | "glass";
  icon?: IconType;
  className?: string;
}) {
  const base = `${LABEL_MD} flex w-full items-center justify-center gap-2 rounded-full px-8 py-4 font-semibold text-white transition-all sm:w-auto`;
  const look =
    variant === "primary"
      ? "border border-white/20 bg-gradient-to-r from-[#7c3aed] via-[#9333ea] to-[#06b6d4] shadow-[inset_0_1px_0_rgba(255,255,255,0.4),0_6px_28px_rgba(124,58,237,0.55)] hover:-translate-y-0.5 hover:shadow-[0_0_36px_rgba(6,182,212,0.7)] active:scale-95"
      : "border border-cyan-500/30 bg-[#0e0d16]/80 shadow-[0_4px_20px_-2px_rgba(0,0,0,0.6),inset_0_1px_0_rgba(255,255,255,0.15)] backdrop-blur-xl hover:border-cyan-400/60 hover:bg-[#1a1829] hover:shadow-[0_0_24px_rgba(6,182,212,0.3)]";
  return (
    <a href={href} className={`${base} ${look} ${className}`}>
      {Icon && <Icon size={18} className={variant === "glass" ? "text-st-tertiary" : ""} />}
      {children}
      {variant === "primary" && <ArrowRight size={18} />}
    </a>
  );
}

const ICON_TONES = [
  "border-st-primary/40 text-st-primary shadow-[inset_0_2px_4px_rgba(0,0,0,0.65),0_0_18px_rgba(124,58,237,0.4)]",
  "border-cyan-400/40 text-cyan-300 shadow-[inset_0_2px_4px_rgba(0,0,0,0.65),0_0_18px_rgba(6,182,212,0.45)]",
  "border-st-secondary/40 text-st-secondary shadow-[inset_0_2px_4px_rgba(0,0,0,0.65),0_0_18px_rgba(206,189,255,0.45)]",
];

/** Glass feature card with a glowing icon tile. */
export function FeatureCard({
  icon: Icon,
  title,
  desc,
  index = 0,
  children,
}: {
  icon?: IconType;
  title: ReactNode;
  desc?: ReactNode;
  index?: number;
  children?: ReactNode;
}) {
  return (
    <Reveal delay={(index % 3) * 0.06} className="st-cosmic-glass group flex h-full flex-col rounded-2xl p-7">
      {Icon && (
        <div className={`mb-5 flex h-12 w-12 items-center justify-center rounded-[3rem] border bg-[#19162e] transition-transform group-hover:scale-105 ${ICON_TONES[index % 3]}`}>
          <Icon size={22} />
        </div>
      )}
      <h3 className="mb-2 font-st-display text-[18px] font-semibold leading-[26px] text-white">{title}</h3>
      {desc && <p className="text-[14px] leading-5 text-slate-300">{desc}</p>}
      {children}
    </Reveal>
  );
}

/** Numbered step cards on a glowing track (the Stitch 4-step lifecycle). */
export function StepTrack({
  steps,
  columns = 4,
}: {
  steps: readonly { title: ReactNode; desc: ReactNode; icon?: IconType }[];
  columns?: 3 | 4;
}) {
  return (
    <div className="relative">
      <div className="absolute left-[10%] right-[10%] top-[44px] z-0 hidden h-[2px] bg-gradient-to-r from-st-primary/30 via-cyan-400/60 to-st-tertiary/70 lg:block">
        <div className="absolute inset-0 bg-gradient-to-r from-transparent via-cyan-300 to-transparent blur-[3px]" />
      </div>
      <ol className={`relative z-10 grid grid-cols-1 gap-6 md:grid-cols-2 ${columns === 3 ? "lg:grid-cols-3" : "lg:grid-cols-4"}`}>
        {steps.map((step, i) => {
          const last = i === steps.length - 1;
          const StepIcon = step.icon;
          return (
            <Reveal
              key={i}
              delay={i * 0.06}
              className={`relative flex flex-col rounded-2xl p-6 ${
                last
                  ? "border border-st-tertiary/40 bg-gradient-to-b from-[#18162e]/95 via-[#110f22]/95 to-[#0b0a16]/95 shadow-[0_20px_48px_-8px_rgba(0,0,0,0.9),0_0_30px_rgba(78,222,163,0.3),inset_0_1px_0_rgba(255,255,255,0.2)] backdrop-blur-2xl"
                  : "st-cosmic-glass"
              }`}
            >
              <div className="mb-5 flex items-center justify-between">
                <div
                  className={`flex h-11 w-11 items-center justify-center rounded-full border font-st-display text-[16px] font-bold ${
                    last
                      ? "border-st-tertiary/60 bg-[#0a2318] text-st-tertiary shadow-[0_0_18px_rgba(78,222,163,0.5)]"
                      : i === 1
                        ? "border-cyan-400/60 bg-[#1a1532] text-cyan-300 shadow-[0_0_18px_rgba(6,182,212,0.5)]"
                        : "border-st-primary/50 bg-[#141224] text-st-primary shadow-[0_0_16px_rgba(124,58,237,0.4)]"
                  }`}
                >
                  {String(i + 1).padStart(2, "0")}
                </div>
                {StepIcon && <StepIcon size={18} className={last ? "text-st-tertiary" : "text-cyan-300"} />}
              </div>
              <h3 className="mb-2 font-st-display text-[18px] font-semibold leading-[26px] text-white">{step.title}</h3>
              <p className="text-[14px] leading-5 text-slate-300">{step.desc}</p>
            </Reveal>
          );
        })}
      </ol>
    </div>
  );
}
