/**
 * Job Search Agent — setup + match status table.
 *
 * Flow: on load, try GET /jobs/setup. 404 → one-time setup (Gemini-suggested
 * designations as editable chips; user can add a totally different role).
 * Otherwise show the saved setup with [Re-setup] and the match table.
 *
 * UI follows the Stitch screen "Jobs Discovery & Matches (Desktop Web)"
 * (project 16808869888425310618, screen d958722f10744021b56e6240152ac8b6) —
 * restyle only (2026-09-24): content, data and behaviour are unchanged; matches
 * render as Stitch requisition cards instead of table rows. Pre-Stitch version:
 * docs/backup/jobs-before-stitch.tsx.txt.
 */
import { useCallback, useEffect, useMemo, useState } from "react";
import { isAxiosError } from "axios";
import { motion } from "framer-motion";
import {
  AlertTriangle,
  ArrowRight,
  BadgeCheck,
  Check,
  CircleCheck,
  FileText,
  Filter,
  Loader2,
  Plus,
  RefreshCw,
  Search,
  Settings2,
  Sparkles,
  Target,
  X,
} from "lucide-react";
import {
  getJobMatches,
  getJobResumes,
  getJobSetup,
  runJobSearch,
  saveJobSetup,
  suggestDesignations,
  updateJobMatchStatus,
  type JobMatch,
  type JobSetup,
  type MatchStatus,
  type ResumeOption,
} from "@/api/jobService";
import { Reveal } from "@/components/motion/reveal";
import { StaggerGroup, StaggerItem } from "@/components/motion/stagger";

type Mode = "loading" | "setup" | "table";

const STATUS_LABELS: Record<MatchStatus, string> = {
  new: "New",
  reviewed: "Reviewed",
  pending_apply: "Pending",
  applied: "Applied",
  dismissed: "Dismissed",
};

const STATUS_BADGE: Record<MatchStatus, string> = {
  new: "bg-st-primary-container/25 text-st-primary",
  reviewed: "bg-st-primary-container/25 text-st-primary",
  pending_apply: "bg-[#ffb95f]/15 text-[#ffb95f]",
  applied: "bg-st-tertiary-container/30 text-st-tertiary",
  dismissed: "bg-st-highest text-st-outline",
};

// Stitch's dark-only styling (see dashboard.tsx for the `st-*` palette).
const RAISED = "shadow-[0_4px_16px_-2px_rgba(0,0,0,0.5),inset_0_1px_0_rgba(255,255,255,0.06)]";
const RECESSED = "shadow-[inset_0_2px_4px_rgba(0,0,0,0.6)]";
const CTA =
  "bg-gradient-to-b from-[#8B5CF6] to-[#6D28D9] text-st-on-primary-container shadow-[inset_0_1px_0_rgba(255,255,255,0.35),0_4px_12px_rgba(124,58,237,0.35)] hover:shadow-[inset_0_1px_0_rgba(255,255,255,0.45),0_6px_16px_rgba(124,58,237,0.55)] active:scale-[0.98] transition-all";
const LABEL_SM = "text-[11px] leading-[14px] tracking-[0.04em] font-semibold";
const LABEL_MD = "text-[13px] leading-4 tracking-[0.02em] font-medium";
const BODY_SM = "text-[12px] leading-[18px] tracking-[0.01em]";
const BODY_LG = "text-[16px] leading-6 tracking-[-0.01em]";
const CODE_SM = "text-[12px] leading-4 font-medium";
const H_SM = "font-st-display text-[18px] leading-[26px] tracking-[-0.01em] font-medium";
const H_MD = "font-st-display text-[24px] leading-8 tracking-[-0.02em] font-semibold";
const EMBLEM_TONES = ["text-st-primary", "text-st-error", "text-st-tertiary", "text-st-secondary"];

const FLOW_STEPS = [
  { title: "Resume & targets", desc: "Pick a resume, confirm the roles" },
  { title: "Agent searches", desc: "Company career pages, ranked" },
  { title: "Review & apply", desc: "Open, apply, track status" },
];

const FILTERS: Array<{ label: string; value: MatchStatus | "all" }> = [
  { label: "All", value: "all" },
  { label: "New", value: "new" },
  { label: "Pending", value: "pending_apply" },
  { label: "Applied", value: "applied" },
  { label: "Dismissed", value: "dismissed" },
];

const errMessage = (err: unknown, fallback: string): string => {
  if (isAxiosError(err)) {
    const detail = (err.response?.data as { detail?: string } | undefined)?.detail;
    if (detail) return detail;
  }
  return err instanceof Error ? err.message : fallback;
};

export default function JobsPage() {
  const [mode, setMode] = useState<Mode>("loading");
  const [setup, setSetup] = useState<JobSetup | null>(null);
  const [matches, setMatches] = useState<JobMatch[]>([]);
  const [filter, setFilter] = useState<MatchStatus | "all">("all");
  const [error, setError] = useState<string | null>(null);
  const [notice, setNotice] = useState<string | null>(null);

  // Setup-mode state
  const [resumes, setResumes] = useState<ResumeOption[]>([]);
  // The job agent uses its OWN resume choice, independent of the interview flow.
  const [resumeId, setResumeId] = useState<number | null>(null);
  const [chips, setChips] = useState<string[]>([]);
  const [chipInput, setChipInput] = useState("");
  const [suggesting, setSuggesting] = useState(false);
  const [saving, setSaving] = useState(false);
  const [searching, setSearching] = useState(false);

  const loadMatches = useCallback(async () => {
    try {
      const data = await getJobMatches();
      setMatches(data.matches);
    } catch (err) {
      setError(errMessage(err, "Failed to load matches"));
    }
  }, []);

  const enterSetupMode = useCallback(async (existing?: JobSetup | null) => {
    setMode("setup");

    // The job side picks its own resume — load the choices, defaulting to the
    // one already saved for jobs (not whatever the interview flow last used).
    try {
      const available = await getJobResumes();
      setResumes(available);
      setResumeId(existing?.resume_document_id ?? available[0]?.id ?? null);
    } catch {
      // Non-fatal: the backend still falls back to the latest resume.
    }

    if (existing?.target_designations?.length) {
      setChips(existing.target_designations);
      return;
    }
    setSuggesting(true);
    try {
      const suggestion = await suggestDesignations();
      setChips(suggestion.designations);
    } catch (err) {
      setError(errMessage(err, "Could not suggest designations — add them manually"));
    } finally {
      setSuggesting(false);
    }
  }, []);

  useEffect(() => {
    const boot = async () => {
      try {
        const existing = await getJobSetup();
        setSetup(existing);
        setMode("table");
        await loadMatches();
      } catch (err) {
        if (isAxiosError(err) && err.response?.status === 404) {
          await enterSetupMode();
        } else {
          setError(errMessage(err, "Failed to load job search"));
          setMode("setup");
        }
      }
    };
    void boot();
  }, [enterSetupMode, loadMatches]);

  const addChip = () => {
    const value = chipInput.trim();
    if (value && !chips.some((c) => c.toLowerCase() === value.toLowerCase())) {
      setChips((prev) => [...prev, value]);
    }
    setChipInput("");
  };

  const onSaveSetup = async () => {
    if (!chips.length) {
      setError("Add at least one target designation");
      return;
    }
    setSaving(true);
    setError(null);
    try {
      const saved = await saveJobSetup({
        target_designations: chips,
        resume_document_id: resumeId,
      });
      setSetup(saved);
      setMode("table");
      await loadMatches();
      setNotice("Setup saved — run a search to find matching jobs");
    } catch (err) {
      setError(errMessage(err, "Failed to save setup"));
    } finally {
      setSaving(false);
    }
  };

  const onSearch = async () => {
    setSearching(true);
    setError(null);
    setNotice(null);
    try {
      const run = await runJobSearch();
      setNotice(
        run.companies_checked === 0
          ? run.message
          : `Checked ${run.companies_checked} companies · ${run.listings_fetched} listings · ${run.new_matches} new matches`
      );
      await loadMatches();
    } catch (err) {
      setError(errMessage(err, "Search failed"));
    } finally {
      setSearching(false);
    }
  };

  const onStatusChange = async (match: JobMatch, status: MatchStatus) => {
    try {
      const updated = await updateJobMatchStatus(match.id, status);
      setMatches((prev) => prev.map((m) => (m.id === updated.id ? updated : m)));
    } catch (err) {
      setError(errMessage(err, "Failed to update status"));
    }
  };

  const visible = useMemo(
    () => (filter === "all" ? matches : matches.filter((m) => m.status === filter)),
    [matches, filter]
  );

  // Which flow step the user is on: setup → search → review.
  const activeStep = mode === "setup" ? 0 : matches.length === 0 ? 1 : 2;

  if (mode === "loading") {
    return (
      <div className="flex min-h-[calc(100vh-3.5rem)] items-center justify-center bg-st-bg">
        <Loader2 className="animate-spin text-st-primary" size={28} />
      </div>
    );
  }

  const stepRail = (
    <div className={`flex flex-col gap-4 rounded-2xl bg-st-low p-6 ${RAISED}`}>
      <div className="flex items-center justify-between">
        <span className={`${H_SM} font-semibold text-st-ink`}>Flow</span>
        <span className={`${LABEL_SM} text-st-outline`}>
          {Math.min(activeStep + 1, FLOW_STEPS.length)} / {FLOW_STEPS.length}
        </span>
      </div>
      <StaggerGroup className="flex flex-col gap-2">
        {FLOW_STEPS.map((step, i) => {
          const state = i === activeStep ? "active" : i < activeStep ? "done" : "todo";
          return (
            <StaggerItem key={step.title}>
              <div
                className={`flex items-center gap-2 rounded-[3rem] p-2 transition-all ${
                  state === "active"
                    ? "bg-st-container shadow-[0_2px_8px_rgba(0,0,0,0.4),inset_0_1px_0_rgba(255,255,255,0.12)]"
                    : `bg-st-lowest ${RECESSED}`
                }`}
              >
                <span
                  className={`${CODE_SM} flex h-8 w-8 shrink-0 items-center justify-center rounded-full font-bold ${
                    state === "todo"
                      ? "bg-st-high text-st-outline"
                      : state === "done"
                        ? "bg-st-tertiary/15 text-st-tertiary"
                        : "bg-st-primary-container text-st-on-primary-container"
                  }`}
                >
                  {state === "done" ? <Check size={16} /> : i + 1}
                </span>
                <span className="min-w-0">
                  <span className={`${LABEL_MD} block truncate font-semibold ${state === "active" ? "text-st-primary" : "text-st-ink"}`}>
                    {step.title}
                  </span>
                  <span className={`${BODY_SM} block truncate text-st-muted`}>{step.desc}</span>
                </span>
              </div>
            </StaggerItem>
          );
        })}
      </StaggerGroup>
    </div>
  );

  return (
    <div className="relative min-h-[calc(100vh-3.5rem)] overflow-hidden bg-st-bg font-st-body text-[14px] leading-5 text-st-ink antialiased">
      {/* Atmospheric underglows */}
      <div className="pointer-events-none absolute left-1/4 top-10 h-96 w-96 rounded-full bg-st-primary-container/10 blur-3xl" />
      <div className="pointer-events-none absolute right-12 top-48 h-80 w-80 rounded-full bg-st-tertiary/5 blur-3xl" />

      <div className="relative mx-auto flex w-full max-w-[90rem] flex-col gap-6 px-4 py-6 sm:px-8">
        {/* Header & status capsule */}
        <motion.section
          initial={{ y: -20, opacity: 0 }}
          animate={{ y: 0, opacity: 1 }}
          className="flex w-full flex-col justify-between gap-4 xl:flex-row xl:items-end"
        >
          <div className="flex min-w-0 max-w-3xl flex-1 flex-col gap-1">
            <span className={`inline-flex w-fit items-center gap-1.5 rounded-full bg-st-lowest px-2 py-0.5 shadow-[inset_0_2px_4px_rgba(0,0,0,0.6)]`}>
              <Sparkles size={12} className="text-st-primary" />
              <span className={`${LABEL_SM} uppercase tracking-widest text-st-primary`}>Job Agent</span>
            </span>
            <h1 className="font-st-display text-[28px] font-bold leading-9 tracking-[-0.02em] text-st-ink md:text-[36px] md:leading-[44px] md:tracking-[-0.025em]">
              Find Your <span className="text-st-primary">Next Role</span>
            </h1>
            <p className={`${BODY_LG} max-w-2xl text-st-muted`}>
              Your resume, matched against live openings on company career pages — ranked, explained,
              and ready to apply.
            </p>
          </div>

          <div className="flex flex-wrap items-center gap-2 rounded-[2rem] bg-st-lowest p-2 shadow-[inset_0_2px_6px_rgba(0,0,0,0.7),0_1px_0_rgba(255,255,255,0.06)] sm:rounded-full xl:shrink-0 xl:flex-nowrap">
            <div className="flex items-center gap-2 rounded-full bg-st-low px-4 py-1 shadow-[0_2px_8px_rgba(0,0,0,0.4),inset_0_1px_0_rgba(255,255,255,0.08)]">
              <BadgeCheck size={18} className="shrink-0 text-st-tertiary" />
              <span className={`${LABEL_MD} text-st-ink`}>
                {mode === "setup"
                  ? "Step 1 — confirm the roles you want to target."
                  : `${matches.length} match${matches.length === 1 ? "" : "es"} found so far.`}
              </span>
            </div>
            {mode === "table" && (
              <div className="flex items-center gap-2">
                <button
                  type="button"
                  onClick={() => void enterSetupMode(setup)}
                  className={`${LABEL_MD} flex items-center gap-1.5 rounded-full bg-st-container px-4 py-1.5 text-st-ink shadow-[inset_0_1px_0_rgba(255,255,255,0.08)] transition-all hover:bg-st-high hover:text-st-primary`}
                >
                  <Settings2 size={16} /> Re-setup
                </button>
                <button
                  type="button"
                  onClick={() => void onSearch()}
                  disabled={searching}
                  className={`${LABEL_MD} ${CTA} flex items-center gap-1.5 rounded-full px-4 py-1.5 font-semibold disabled:opacity-60`}
                >
                  {searching ? <Loader2 size={16} className="animate-spin" /> : <Search size={16} />}
                  {searching ? "Searching…" : "Search Now"}
                </button>
              </div>
            )}
          </div>
        </motion.section>

        {error && (
          <p className={`${BODY_SM} rounded-2xl bg-[#93000a]/25 px-4 py-3 text-st-error shadow-[inset_0_1px_0_rgba(255,180,171,0.15)]`}>
            {error}
          </p>
        )}
        {notice && (
          <p className={`${BODY_SM} rounded-2xl bg-st-tertiary/10 px-4 py-3 text-st-tertiary shadow-[inset_0_1px_0_rgba(78,222,163,0.2)]`}>
            {notice}
          </p>
        )}

        {mode === "table" && (
          /* Query bar: current targets + status filters */
          <section className={`flex w-full flex-col items-stretch gap-4 rounded-2xl bg-st-low p-4 lg:flex-row lg:items-center ${RAISED}`}>
            <div className="flex min-w-0 flex-1 items-center gap-2 rounded-full bg-st-lowest px-4 py-2 shadow-[inset_0_2px_4px_rgba(0,0,0,0.7),0_1px_0_rgba(255,255,255,0.04)]">
              <Filter size={18} className="shrink-0 text-st-outline" />
              <span className="truncate text-st-muted">
                Targeting:{" "}
                <span className="font-medium text-st-ink">{setup?.target_designations.join(" · ") || "—"}</span>
              </span>
            </div>
            <div className="flex items-center gap-1 overflow-x-auto pb-1 lg:pb-0">
              {FILTERS.map((f) => {
                const active = filter === f.value;
                return (
                  <button
                    key={f.value}
                    type="button"
                    onClick={() => setFilter(f.value)}
                    aria-pressed={active}
                    className={`${LABEL_MD} flex items-center gap-1.5 whitespace-nowrap rounded-full px-4 py-1 transition-all ${
                      active
                        ? "bg-st-high text-st-primary shadow-[inset_0_1px_0_rgba(255,255,255,0.1),0_2px_6px_rgba(0,0,0,0.4)]"
                        : "bg-st-lowest text-st-muted shadow-[inset_0_2px_4px_rgba(0,0,0,0.6)] hover:text-st-ink"
                    }`}
                  >
                    {active && <span className="h-1.5 w-1.5 rounded-full bg-st-primary" />}
                    {f.label}
                  </button>
                );
              })}
              <button
                type="button"
                onClick={() => void loadMatches()}
                aria-label="Refresh matches"
                className="ml-1 rounded-full bg-st-lowest p-2 text-st-muted shadow-[inset_0_2px_4px_rgba(0,0,0,0.6)] transition-colors hover:text-st-primary"
              >
                <RefreshCw size={16} />
              </button>
            </div>
          </section>
        )}

        {/* Master-detail surface */}
        <section className="grid w-full grid-cols-1 items-start gap-6 xl:grid-cols-12">
          <aside className="flex flex-col gap-4 xl:col-span-3">
            {stepRail}

            {mode === "table" && setup && (
              <div className={`flex flex-col gap-4 rounded-2xl bg-st-low p-6 ${RAISED}`}>
                <div className="flex items-center justify-between">
                  <span className={`${H_SM} font-semibold text-st-ink`}>Target designations</span>
                  <Target size={16} className="text-st-outline" />
                </div>
                <div className="flex flex-col gap-2">
                  {setup.target_designations.length === 0 && <span className={`${BODY_SM} text-st-outline`}>—</span>}
                  {setup.target_designations.map((d) => (
                    <div key={d} className={`flex items-center gap-2 rounded-[3rem] bg-st-lowest px-3 py-2 ${RECESSED}`}>
                      <span className="h-2 w-2 shrink-0 rounded-full bg-st-tertiary" />
                      <span className="truncate font-medium text-st-ink">{d}</span>
                    </div>
                  ))}
                </div>
              </div>
            )}
          </aside>

          <div className="flex flex-col gap-6 xl:col-span-9">
            {mode === "setup" && (
              <Reveal className="flex flex-col gap-4">
                {/* Step 1a — the job side's OWN resume, separate from the interview's */}
                <div className={`flex flex-col gap-4 rounded-2xl bg-st-low p-6 ${RAISED}`}>
                  <div className="flex items-start gap-4">
                    <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-[3rem] bg-st-lowest shadow-[inset_0_2px_4px_rgba(0,0,0,0.8),0_1px_0_rgba(255,255,255,0.08)]">
                      <FileText size={20} className="text-st-primary" />
                    </div>
                    <div>
                      <h2 className={`${H_MD} text-st-ink`}>Resume for job search</h2>
                      <p className={`${BODY_SM} text-st-muted`}>
                        Choose which resume the agent matches jobs against. This is separate from the
                        resume your mock interviews use — you can point each side at a different one.
                      </p>
                    </div>
                  </div>

                  {resumes.length === 0 ? (
                    <div className={`rounded-[2rem] bg-st-lowest p-4 ${RECESSED}`}>
                      <p className={`${BODY_SM} text-st-muted`}>
                        No resume indexed yet. Upload one in the interview flow and it becomes available
                        here too.
                      </p>
                      <a
                        href="/interview/select-role"
                        className={`${LABEL_MD} mt-2 inline-flex items-center gap-1 font-semibold text-st-primary hover:underline`}
                      >
                        Upload a resume →
                      </a>
                    </div>
                  ) : (
                    <div className="grid gap-3 sm:grid-cols-2">
                      {resumes.map((r) => {
                        const selected = r.id === resumeId;
                        return (
                          <button
                            key={r.id}
                            type="button"
                            onClick={() => setResumeId(r.id)}
                            aria-pressed={selected}
                            className={`rounded-[2rem] p-4 text-left transition-all ${
                              selected
                                ? "bg-st-container shadow-[0_0_0_1px_#7c3aed,0_0_14px_rgba(124,58,237,0.3),inset_0_1px_0_rgba(255,255,255,0.12)]"
                                : `bg-st-lowest hover:bg-st-container ${RECESSED}`
                            }`}
                          >
                            <div className="flex items-start justify-between gap-2">
                              <span className={`${LABEL_MD} truncate font-semibold text-st-ink`}>{r.file_name}</span>
                              {selected && <Check size={16} className="shrink-0 text-st-tertiary" />}
                            </div>
                            <p className={`${BODY_SM} mt-1 truncate text-st-muted`}>
                              {[r.role, r.experience].filter(Boolean).join(" · ") || "Indexed resume"}
                            </p>
                            {r.skills.length > 0 && (
                              <p className={`${CODE_SM} mt-2 truncate text-st-outline`}>{r.skills.slice(0, 4).join(", ")}</p>
                            )}
                          </button>
                        );
                      })}
                    </div>
                  )}
                </div>

                <div className={`flex flex-col gap-4 rounded-2xl bg-st-low p-6 ${RAISED}`}>
                  <div className="flex items-start gap-4">
                    <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-[3rem] bg-st-lowest shadow-[inset_0_2px_4px_rgba(0,0,0,0.8),0_1px_0_rgba(255,255,255,0.08)]">
                      <Target size={20} className="text-st-tertiary" />
                    </div>
                    <div>
                      <h2 className={`${H_MD} text-st-ink`}>Target designations</h2>
                      <p className={`${BODY_SM} text-st-muted`}>
                        Suggested from your resume — remove any, or add a different role you want to target
                        (e.g. switch from Python to Frontend). One resume can target many roles.
                      </p>
                    </div>
                  </div>

                  {suggesting ? (
                    <p className={`${BODY_SM} flex items-center gap-2 text-st-muted`}>
                      <Loader2 size={16} className="animate-spin" /> Analyzing your resume…
                    </p>
                  ) : (
                    <div className="flex flex-wrap gap-2">
                      {chips.map((chip) => (
                        <span
                          key={chip}
                          className={`${CODE_SM} flex items-center gap-1 rounded-full bg-st-lowest py-1 pl-3 pr-1 text-st-ink shadow-[inset_0_2px_4px_rgba(0,0,0,0.6)]`}
                        >
                          {chip}
                          <button
                            type="button"
                            aria-label={`Remove ${chip}`}
                            onClick={() => setChips((prev) => prev.filter((c) => c !== chip))}
                            className="rounded-full p-0.5 text-st-outline transition-colors hover:bg-st-high hover:text-st-error"
                          >
                            <X size={14} />
                          </button>
                        </span>
                      ))}
                      {!chips.length && (
                        <span className={`${BODY_SM} text-st-outline`}>No designations yet — add one below</span>
                      )}
                    </div>
                  )}

                  <div className="flex flex-wrap gap-2">
                    <div className="flex min-w-0 max-w-md flex-1 items-center gap-2 rounded-full bg-st-lowest px-4 py-2 shadow-[inset_0_2px_4px_rgba(0,0,0,0.7),0_1px_0_rgba(255,255,255,0.04)] transition-all focus-within:shadow-[inset_0_2px_4px_rgba(0,0,0,0.7),0_0_0_1px_#7C3AED,0_0_14px_rgba(124,58,237,0.3)]">
                      <Plus size={16} className="shrink-0 text-st-outline" />
                      <input
                        value={chipInput}
                        onChange={(e) => setChipInput(e.target.value)}
                        onKeyDown={(e) => {
                          if (e.key === "Enter") {
                            e.preventDefault();
                            addChip();
                          }
                        }}
                        placeholder="Add a designation (e.g. Frontend Developer)"
                        className="w-full border-0 bg-transparent text-st-ink outline-none placeholder:text-st-outline"
                      />
                    </div>
                    <button
                      type="button"
                      onClick={addChip}
                      className={`${LABEL_MD} flex items-center gap-1 rounded-full bg-st-container px-4 py-2 text-st-ink shadow-[inset_0_1px_0_rgba(255,255,255,0.08)] transition-all hover:bg-st-high hover:text-st-primary`}
                    >
                      <Plus size={16} /> Add
                    </button>
                  </div>

                  <div className="flex justify-end border-t border-white/5 pt-4">
                    <button
                      type="button"
                      onClick={() => void onSaveSetup()}
                      disabled={saving || suggesting}
                      className={`${LABEL_MD} ${CTA} rounded-full px-6 py-2 font-semibold disabled:opacity-60`}
                    >
                      {saving ? "Saving…" : "Save & Continue"}
                    </button>
                  </div>
                </div>
              </Reveal>
            )}

            {mode === "table" && (
              <div className="flex flex-col gap-4">
                {visible.length === 0 && (
                  <div className={`rounded-2xl bg-st-low p-8 text-center text-st-muted ${RAISED}`}>
                    {matches.length === 0
                      ? "No matches yet — click Search Now to scan career pages"
                      : "No matches with this status"}
                  </div>
                )}

                {visible.map((m, idx) => {
                  const top = idx === 0;
                  return (
                    <Reveal
                      key={m.id}
                      delay={Math.min(idx, 6) * 0.04}
                      className={`relative overflow-hidden rounded-2xl bg-st-low transition-all duration-200 ${
                        top
                          ? "shadow-[0_16px_32px_-6px_rgba(0,0,0,0.7),0_0_20px_-2px_rgba(124,58,237,0.25),inset_0_1px_0_rgba(255,255,255,0.12)]"
                          : RAISED
                      } ${m.status === "dismissed" ? "opacity-60" : ""}`}
                    >
                      {top && <div className="h-1 w-full bg-gradient-to-r from-st-primary-container via-st-primary to-st-tertiary" />}
                      <div className="flex flex-col gap-4 p-6">
                        <div className="flex flex-col justify-between gap-4 lg:flex-row lg:items-center">
                          <div className="flex min-w-0 items-start gap-4">
                            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-[3rem] bg-st-lowest shadow-[inset_0_2px_4px_rgba(0,0,0,0.8),0_1px_0_rgba(255,255,255,0.08)]">
                              <span className={`font-st-display text-[18px] font-bold ${EMBLEM_TONES[idx % EMBLEM_TONES.length]}`}>
                                {m.company.trim().charAt(0).toUpperCase() || "?"}
                              </span>
                            </div>
                            <div className="flex min-w-0 flex-col">
                              <div className="flex flex-wrap items-center gap-2">
                                <h2 className={`${H_MD} font-bold text-st-ink`}>{m.company}</h2>
                                <span
                                  className={`${LABEL_SM} inline-flex items-center gap-1 rounded-full px-2 py-0.5 ${STATUS_BADGE[m.status]}`}
                                >
                                  <span className="h-1.5 w-1.5 rounded-full bg-current" />
                                  {STATUS_LABELS[m.status]}
                                </span>
                              </div>
                              <a
                                href={m.url}
                                target="_blank"
                                rel="noreferrer"
                                className={`${H_SM} font-semibold text-st-secondary hover:underline`}
                              >
                                {m.title}
                              </a>
                              <span className={`${BODY_SM} text-st-muted`}>
                                {m.location || "—"}
                                {m.remote ? " (Remote)" : ""}
                              </span>
                            </div>
                          </div>

                          <div className="flex items-center justify-between gap-1 lg:flex-col lg:items-end">
                            <div className="flex items-baseline gap-1 rounded-2xl bg-st-lowest px-4 py-1 shadow-[inset_0_2px_4px_rgba(0,0,0,0.8),0_1px_0_rgba(255,255,255,0.04)]">
                              <span className="font-st-display text-[40px] font-bold leading-[48px] tracking-[-0.03em] text-st-primary">
                                {Math.round(m.match_score)}
                              </span>
                              <span className={`${H_SM} font-bold text-[#eaddff]`}>%</span>
                            </div>
                            <span className={`${LABEL_SM} uppercase tracking-wider text-st-outline`}>Match</span>
                          </div>
                        </div>

                        {m.match_reasons?.fits?.length ? (
                          <div className="flex flex-wrap items-center gap-1">
                            {m.match_reasons.fits.map((fit) => (
                              <span
                                key={fit}
                                className={`${CODE_SM} rounded-full bg-st-lowest px-2 py-1 text-st-ink shadow-[inset_0_2px_4px_rgba(0,0,0,0.6)]`}
                              >
                                {fit}
                              </span>
                            ))}
                          </div>
                        ) : null}

                        <div className="flex flex-col justify-between gap-3 border-t border-white/5 pt-4 sm:flex-row sm:items-center">
                          <div className={`${BODY_SM} flex min-w-0 items-center gap-2`}>
                            {m.status === "pending_apply" && m.pending_reason ? (
                              <>
                                <AlertTriangle size={16} className="shrink-0 text-[#ffb95f]" />
                                <span className="text-[#ffb95f]">{m.pending_reason}</span>
                              </>
                            ) : null}
                          </div>
                          <div className="flex flex-wrap items-center gap-2">
                            {m.status !== "dismissed" && (
                              <button
                                type="button"
                                onClick={() => void onStatusChange(m, "dismissed")}
                                className={`${LABEL_MD} rounded-full bg-st-lowest px-4 py-1.5 text-st-muted shadow-[inset_0_1px_2px_rgba(0,0,0,0.6)] transition-all hover:text-st-ink`}
                              >
                                Dismiss
                              </button>
                            )}
                            {m.status !== "applied" && (
                              <button
                                type="button"
                                onClick={() => void onStatusChange(m, "applied")}
                                className={`${LABEL_MD} flex items-center gap-1 rounded-full bg-st-container px-4 py-1.5 text-st-tertiary shadow-[inset_0_1px_0_rgba(255,255,255,0.08)] transition-all hover:bg-st-high`}
                              >
                                <CircleCheck size={14} /> Mark Applied
                              </button>
                            )}
                            <a
                              href={m.apply_url || m.url}
                              target="_blank"
                              rel="noreferrer"
                              onClick={() => {
                                if (m.status === "new") void onStatusChange(m, "reviewed");
                              }}
                              className={`${LABEL_MD} ${CTA} flex items-center gap-1 rounded-full px-5 py-1.5 font-semibold`}
                            >
                              Apply <ArrowRight size={14} />
                            </a>
                          </div>
                        </div>
                      </div>
                    </Reveal>
                  );
                })}
              </div>
            )}
          </div>
        </section>
      </div>
    </div>
  );
}
